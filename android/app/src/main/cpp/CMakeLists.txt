cmake_minimum_required(VERSION 3.22)
project(dictation_whisper LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL ON CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
if(GGML_VULKAN)
  find_program(HOST_GLSLC NAMES glslc NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  find_program(HOST_C_COMPILER NAMES gcc clang NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  find_program(HOST_CXX_COMPILER NAMES g++ clang++ NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  find_path(HOST_VULKAN_INCLUDE_DIR NAMES vulkan/vulkan.hpp
    PATHS /usr/local/include /usr/include NO_CMAKE_FIND_ROOT_PATH REQUIRED)
  # Isolate Vulkan headers so host libc headers never enter Android's include path.
  set(VULKAN_HEADERS_DIR "${CMAKE_CURRENT_BINARY_DIR}/vulkan-headers")
  file(MAKE_DIRECTORY "${VULKAN_HEADERS_DIR}")
  file(COPY "${HOST_VULKAN_INCLUDE_DIR}/vulkan" DESTINATION "${VULKAN_HEADERS_DIR}")
  if(EXISTS "${HOST_VULKAN_INCLUDE_DIR}/vk_video")
    file(COPY "${HOST_VULKAN_INCLUDE_DIR}/vk_video" DESTINATION "${VULKAN_HEADERS_DIR}")
  endif()
  set(Vulkan_GLSLC_EXECUTABLE "${HOST_GLSLC}" CACHE FILEPATH "" FORCE)
  set(Vulkan_INCLUDE_DIR "${VULKAN_HEADERS_DIR}" CACHE PATH "" FORCE)
  if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(VULKAN_ARCH aarch64-linux-android)
  elseif(ANDROID_ABI STREQUAL "x86_64")
    set(VULKAN_ARCH x86_64-linux-android)
  else()
    message(FATAL_ERROR "Unsupported Vulkan ABI: ${ANDROID_ABI}")
  endif()
  # Vulkan is loaded as an optional backend; its NDK stub starts at API 24.
  set(Vulkan_LIBRARY "${CMAKE_SYSROOT}/usr/lib/${VULKAN_ARCH}/24/libvulkan.so" CACHE FILEPATH "" FORCE)
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/whisper.cpp/CMakeLists.txt")
  message(FATAL_ERROR "Run npm run android:prepare-whisper before building Android")
endif()
add_subdirectory(vendor/whisper.cpp whisper)
add_library(dictation-whisper SHARED whisper-jni.cpp)
target_link_libraries(dictation-whisper PRIVATE whisper android log)
if(GGML_VULKAN)
  target_compile_definitions(dictation-whisper PRIVATE DICTATION_VULKAN)
endif()
